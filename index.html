    <script>
        let cartCount = 0;
        let orders = [];

        function switchView(view) {
            const menuView = document.getElementById('view-menu');
            const cashierView = document.getElementById('view-cashier');
            const btnMenu = document.getElementById('btn-menu');
            const btnCashier = document.getElementById('btn-cashier');

            if (view === 'menu') {
                menuView.classList.remove('hidden');
                cashierView.classList.add('hidden');
                btnMenu.className = 'btn';
                btnCashier.className = 'btn-sec';
            } else {
                menuView.classList.add('hidden');
                cashierView.classList.remove('hidden');
                btnCashier.className = 'btn';
                btnMenu.className = 'btn-sec';
            }
        }

        function addToCart(itemName, price) {
            cartCount++;
            document.getElementById('badge-count').innerText = cartCount;

            orders.push({ name: itemName, price: price });
            updateCashierScreen();
            alert('تمت إضافة ' + itemName + ' إلى الطلب بنجاح!');
        }

        function updateCashierScreen() {
            const listDiv = document.getElementById('orders-list');
            if (orders.length === 0) {
                listDiv.innerHTML = 'لا توجد طلبات جديدة حالياً.';
                return;
            }

            let html = '';
            orders.forEach((order, index) => {
                html += `<div style="display: flex; justify-content: space-between; align-items: center; border-bottom: 1px solid #334155; padding: 8px 0; text-align: right;">
                    <span>${index + 1}. ${order.name} (${order.price} د.ع)</span>
                    <span style="color: #10b981; font-weight: bold;">قيد التجهيز</span>
                </div>`;
            });
            listDiv.innerHTML = html;
        }
